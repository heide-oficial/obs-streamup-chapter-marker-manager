cmake_minimum_required(VERSION 3.16)
project(chapter-state-tests LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Test)
add_executable(chapter-state-tests chapter-state-tests.cpp)
target_link_libraries(chapter-state-tests PRIVATE Qt6::Core)
target_compile_features(chapter-state-tests PRIVATE cxx_std_17)
enable_testing()
add_test(NAME chapter-state COMMAND chapter-state-tests)
add_executable(chapter-picker-tests chapter-picker-tests.cpp)
target_link_libraries(chapter-picker-tests PRIVATE Qt6::Widgets Qt6::Test)
target_compile_features(chapter-picker-tests PRIVATE cxx_std_17)
add_test(NAME chapter-picker COMMAND chapter-picker-tests)
set_tests_properties(chapter-picker PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
